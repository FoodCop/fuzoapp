  {emoji:'❤️', title:'Community Favorite', stat:'likesReceived', target:1000},
  {emoji:'🤝', title:'Food Companion', stat:null, target:null, future:true}
];

const XP_PER_LEVEL = 300;
const ROMAN = ['0','I','II','III','IV','V','VI'];

/* ========================= STATE ========================= */
function initialStats(){
  return {
    restaurantsVisited:0, placesSaved:0, locationsPinned:0, cuisinesExplored:0,
    reviewsWritten:0, photosUploaded:0, videosUploaded:0, recipesUploaded:0,
    likesReceived:0, followers:0, helpfulVotes:0, bookmarksReceived:0, followsGiven:0,
    neighborhoodsExplored:0, hiddenGemsVisited:0, citiesExplored:0, countriesVisited:0,
    kmTraveled:0, monthsActive:0, posts:0, profileVisits:0, likesOnPosts:0,
    travelStoriesPosted:0, tutorialsPublished:0, reviewAvgLength:0,
    cafesVisited:0, pizzaPlaces:0, sushiTimes:0, burgerVisits:0, spicyReviews:0,
    dessertsShared:0, veganVisits:0, streetFoodCheckins:0,
    tasteProfileComplete:false, topReviewerInCity:false,
    invitationOrQualityBlogger:false, communityRecognitionTravel:false,
    highEngagementRecipe:false, invitationOrQualityInstructor:false
  };
}
function initialState(){
  return { xp:0, stats:initialStats(), activeRole:'explorer', earnedBefore:{}, simOpen:false };
}
let state = initialState();

/* ========================= HELPERS ========================= */
function escapeHtml(s){ return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }
function clamp01(n){ return Math.max(0, Math.min(1, n)); }

/* ========================= PROGRESSION LOGIC ========================= */
function getLevel(xp){ return Math.floor(xp / XP_PER_LEVEL) + 1; }
function getLevelProgress(xp){ return (xp % XP_PER_LEVEL) / XP_PER_LEVEL; }

function reqProgress(req){
  const val = state.stats[req.stat];
  if(req.target === true) return val === true ? 1 : 0;
  return clamp01((val || 0) / req.target);
}
function reqDone(req){ return reqProgress(req) >= 1; }
function badgeProgress(badge){
  const sum = badge.reqs.reduce((s,r)=>s+reqProgress(r), 0);
  return sum / badge.reqs.length;
}
function isBadgeEarned(badge){ return badge.reqs.every(reqDone); }
function roleEarnedCount(role){ return role.badges.filter(isBadgeEarned).length; }
function roleRankLabel(role){
  const count = roleEarnedCount(role);
  return count === 0 ? `${role.rankName} \u2014 Unranked` : `${role.rankName} ${ROMAN[count]}`;
}

function allEarnedKeys(){
  const keys = {};
  ROLES.forEach(role=>{
    role.badges.forEach((b,i)=>{ if(isBadgeEarned(b)) keys[role.key+':'+i] = true; });
  });
  ACHIEVEMENTS.forEach((a,i)=>{
    if(!a.future && (state.stats[a.stat]||0) >= a.target) keys['ach:'+i] = true;
  });
  return keys;
}

/* ========================= CELEBRATION ========================= */
function celebrate(title, rewardText){
  const root = document.getElementById('celebrate-root');
  const el = document.createElement('div');
  el.className = 'celebrate';
  el.innerHTML = `<span class="celebrate__emoji">\u{1F389}</span>${escapeHtml(title)} unlocked!<br /><span class="faint">${escapeHtml(rewardText||'')}</span>`;
  root.appendChild(el);
  requestAnimationFrame(()=> el.classList.add('celebrate--in'));
  setTimeout(()=>{ el.classList.remove('celebrate--in'); }, 2600);
  setTimeout(()=> el.remove(), 3000);
}
function checkNewUnlocks(){
  const now = allEarnedKeys();
  const newlyEarned = [];
  Object.keys(now).forEach(k=>{
    if(!state.earnedBefore[k]){
      if(k.indexOf('ach:')===0){
        const a = ACHIEVEMENTS[parseInt(k.split(':')[1],10)];
        newlyEarned.push({title:a.title, reward:'Achievement unlocked'});
      } else {
        const [roleKey, idx] = k.split(':');
        const role = ROLES.find(r=>r.key===roleKey);
        const b = role.badges[parseInt(idx,10)];
        newlyEarned.push({title:b.title, reward:b.reward});
      }
    }
  });
  state.earnedBefore = now;
  if(newlyEarned.length){
    newlyEarned.forEach((n,i)=> setTimeout(()=>celebrate(n.title, n.reward), i*900));
  }
}

/* ========================= ACTIONS ========================= */
function applyEffect(effect){
  Object.keys(effect).forEach(k=>{ state.stats[k] = (state.stats[k]||0) + effect[k]; });
}
function simulateXpAction(key){
  const action = XP_ACTIONS.find(a=>a.key===key);
  if(!action) return;
  state.xp += action.xp;
  applyEffect(action.effect);
  checkNewUnlocks();
  render();
}
function simulateOtherAction(key){
  const action = OTHER_ACTIONS.find(a=>a.key===key);
  if(!action) return;
  applyEffect(action.effect);
  checkNewUnlocks();
  render();
}
function toggleManualFlag(key){
  state.stats[key] = !state.stats[key];
  checkNewUnlocks();
  render();
}
function setActiveRole(key){ state.activeRole = key; render(); }
function toggleSim(){ state.simOpen = !state.simOpen; render(); }
function doReset(){ state = initialState(); render(); }

/* ========================= RING SVG ========================= */
function ringSvg(fraction, size, stroke, color, trackColor){
  const r = (size - stroke)/2, c = 2*Math.PI*r;
  const offset = c*(1-clamp01(fraction));
  const center = size/2;
  return `<svg width="${size}" height="${size}" viewBox="0 0 ${size} ${size}">
    <circle cx="${center}" cy="${center}" r="${r}" stroke="${trackColor||'var(--track)'}" stroke-width="${stroke}" fill="none" />
    <circle cx="${center}" cy="${center}" r="${r}" stroke="${color}" stroke-width="${stroke}" fill="none"
      stroke-dasharray="${c}" stroke-dashoffset="${offset}" stroke-linecap="round"
      transform="rotate(-90 ${center} ${center})" style="transition:stroke-dashoffset .4s ease" />
  </svg>`;
}

/* ========================= RENDER: HEADER ========================= */
function renderHeader(){
  const xp = state.xp, level = getLevel(xp), pct = getLevelProgress(xp);
  const role = ROLES.find(r=>r.key===state.activeRole);
  return `<div class="level-card">
    <div class="level-ring">${ringSvg(pct, 84, 9, 'var(--gold)')}
      <div class="level-ring__face"><div class="level-ring__num">${level}</div><div class="level-ring__tag">LEVEL</div></div>
    </div>
    <div class="level-info">
      <div class="level-info__name">Your FUZO Progress</div>
      <div class="level-info__rank">${escapeHtml(roleRankLabel(role))}</div>
      <div class="level-info__xp">Taste Score: ${xp.toLocaleString()} XP \u00b7 ${(xp % XP_PER_LEVEL)}/${XP_PER_LEVEL} to next level</div>
      <div class="xp-track"><div class="xp-track__fill" style="width:${pct*100}%"></div></div>
    </div>
  </div>`;
}

/* ========================= RENDER: ROLE SWITCHER ========================= */
function renderRoleRow(){
  return `<div class="role-row">${ROLES.map(r=>`
    <button type="button" class="role-chip ${state.activeRole===r.key?'role-chip--on':''}"
      style="--role-accent:${r.accent}; --role-accent-dark:${r.accentDark}"
      data-action="set-role" data-role="${r.key}">${escapeHtml(r.label)}</button>
  `).join('')}</div>`;
}

/* ========================= RENDER: BADGE TRACK ========================= */
function renderBadgeTrack(){
  const role = ROLES.find(r=>r.key===state.activeRole);
  const earned = roleEarnedCount(role);
  const cards = role.badges.map((b,i)=>{
    const done = isBadgeEarned(b);
    const reqsHtml = b.reqs.map(r=>{
      const p = reqProgress(r);
      const isBool = r.target === true;
      const valText = isBool ? (state.stats[r.stat] ? 'Done' : 'Not yet') : `${Math.min(state.stats[r.stat]||0, r.target)}/${r.target}`;
      let manualBtn = '';
      if(isBool && MANUAL_FLAGS.some(f=>f.key===r.stat)){
        manualBtn = `<button type="button" class="manual-toggle ${state.stats[r.stat]?'manual-toggle--on':''}" data-action="toggle-flag" data-flag="${r.stat}">
          <span class="manual-toggle__dot"></span>${state.stats[r.stat] ? 'Marked complete' : 'Mark complete (demo)'}</button>`;
      }
      return `<div class="req ${p>=1?'req--done':''}"><span>${p>=1?'\u2713 ':''}${escapeHtml(r.label)}</span><span>${valText}</span></div>
        <div class="req__bar" style="--role-accent:${role.accent}; --role-accent-dark:${role.accentDark}"><div class="req__fill" style="width:${p*100}%; --role-accent:${role.accent}; --role-accent-dark:${role.accentDark}"></div></div>
        ${manualBtn}`;
    }).join('');
    return `<div class="badge-card ${done?'badge-card--earned':'badge-card--locked'}">
      <div class="badge-card__emoji-wrap">${b.emoji}</div>
      <div class="badge-card__body">
        <div class="badge-card__top"><span class="badge-card__title">${i+1}. ${escapeHtml(b.title)}</span>
          <span class="badge-card__state ${done?'badge-card__state--earned':'badge-card__state--locked'}">${done?'Earned \u2713':'Locked'}</span>
        </div>
        ${reqsHtml}
        <div class="badge-card__reward">\u{1F381} Reward: ${escapeHtml(b.reward)}</div>
      </div>
    </div>`;
  }).join('');
  return `<div class="section">
    <div class="section__head"><span class="h2">${escapeHtml(role.label)} Badges</span><span class="faint">${earned}/6 earned</span></div>
    ${cards}
  </div>`;
}

/* ========================= RENDER: ACHIEVEMENTS ========================= */
function renderAchievements(){
  const cards = ACHIEVEMENTS.map(a=>{
    if(a.future){
      return `<div class="ach-card ach-card--future">
        <div class="ach-ring">${ringSvg(0, 50, 6, 'var(--lock)')}<span class="ach-ring__emoji">${a.emoji}</span></div>
        <div class="ach-card__title">${escapeHtml(a.title)}</div>
        <div class="ach-card__sub">Coming soon</div>
      </div>`;
    }
    const val = state.stats[a.stat]||0;
    const p = clamp01(val/a.target);
    const done = val >= a.target;
    return `<div class="ach-card ${done?'':'ach-card--locked'}">
      <div class="ach-ring">${ringSvg(p, 50, 6, done?'#4C7A1E':'var(--accent)')}<span class="ach-ring__emoji">${a.emoji}</span></div>
      <div class="ach-card__title">${escapeHtml(a.title)}</div>
      <div class="ach-card__sub">${done? 'Unlocked \u2713' : Math.min(val,a.target)+'/'+a.target}</div>
    </div>`;
  }).join('');
  const earnedCount = ACHIEVEMENTS.filter(a=>!a.future && (state.stats[a.stat]||0)>=a.target).length;
  return `<div class="section">
    <div class="section__head"><span class="h2">Achievement Milestones</span><span class="faint">${earnedCount}/${ACHIEVEMENTS.length-1} unlocked</span></div>
    <div class="ach-grid">${cards}</div>
  </div>`;
}

/* ========================= RENDER: SIMULATOR ========================= */
function renderSimulator(){
  if(!state.simOpen){
    return `<button type="button" class="sim-toggle" data-action="toggle-sim"><span>\u{1F9EA} Demo: simulate activity</span><span>+</span></button>`;
  }
  const xpBtns = XP_ACTIONS.map(a=>`<button type="button" class="sim-btn sim-btn--xp" data-action="sim-xp" data-key="${a.key}">${escapeHtml(a.label)} (+${a.xp}xp)</button>`).join('');
  const otherBtns = OTHER_ACTIONS.map(a=>`<button type="button" class="sim-btn" data-action="sim-other" data-key="${a.key}">${escapeHtml(a.label)}</button>`).join('');
  const flagBtns = MANUAL_FLAGS.map(f=>`<button type="button" class="sim-btn ${state.stats[f.key]?'sim-btn--xp':''}" data-action="toggle-flag" data-flag="${f.key}">${escapeHtml(f.label)}</button>`).join('');
  return `<button type="button" class="sim-toggle" data-action="toggle-sim"><span>\u{1F9EA} Demo: simulate activity</span><span>\u2212</span></button>
    <div class="sim-panel">
      <div class="sim-group"><div class="sim-group__label">XP actions</div><div class="sim-btns">${xpBtns}</div></div>
      <div class="sim-group"><div class="sim-group__label">Other activity (no XP, feeds badges)</div><div class="sim-btns">${otherBtns}</div></div>
      <div class="sim-group"><div class="sim-group__label">Manual / judgment flags (demo only)</div><div class="sim-btns">${flagBtns}</div></div>
    </div>`;
}

/* ========================= ROOT RENDER ========================= */
function render(){
  document.getElementById('app').innerHTML = `<div class="stage"><div class="menu">
    <span class="brandmark">FUZO</span>
    <div class="topbar"><button type="button" class="iconbtn" data-action="noop-profile">\u2190 Profile</button><span></span></div>
    <div class="content">
      <h1 class="h1">Rewards & Progression</h1>
      <p class="muted">Your universal level grows from everything you do. Your rank badge grows from your chosen role.</p>
      ${renderHeader()}
      ${renderRoleRow()}
      ${renderBadgeTrack()}
      ${renderAchievements()}
      ${renderSimulator()}
    </div>
    <div class="reset-row"><button type="button" class="linkbtn" data-action="reset">\u21ba Reset demo progress</button></div>
  </div></div>`;
}

/* ========================= EVENTS ========================= */
document.getElementById('app').addEventListener('click', function(e){
  const btn = e.target.closest('[data-action]');
  if(!btn) return;
  switch(btn.dataset.action){
    case 'set-role': setActiveRole(btn.dataset.role); break;
    case 'sim-xp': simulateXpAction(btn.dataset.key); break;
    case 'sim-other': simulateOtherAction(btn.dataset.key); break;
    case 'toggle-flag': toggleManualFlag(btn.dataset.flag); break;
    case 'toggle-sim': toggleSim(); break;
    case 'reset': doReset(); break;
    case 'noop-profile': break;
  }
});

render();
</script>
