/* ========================= INTERACTION HANDLERS ========================= */
function toggleChip(btn){
  const group = btn.dataset.group, value = btn.dataset.value;
  const maxAttr = btn.dataset.max; const max = maxAttr ? parseInt(maxAttr,10) : null;
  const arr = getByPath(state, group);
  const idx = arr.indexOf(value);
  if(idx>=0){ arr.splice(idx,1); }
  else{
    if(max && arr.length>=max) return;
    if(group==='a.dietary'){
      if(value==='No Restrictions'){ arr.length = 0; }
      else{ const nr = arr.indexOf('No Restrictions'); if(nr>=0) arr.splice(nr,1); }
    }
    arr.push(value);
  }
  render();
}
function addCustom(btn){
  const group = btn.dataset.group, extra = btn.dataset.extra, inputId = btn.dataset.input;
  const maxAttr = btn.dataset.max; const max = maxAttr ? parseInt(maxAttr,10) : null;
  const input = document.getElementById(inputId);
  if(!input) return;
  const val = input.value.trim();
  input.value = '';
  if(!val){ render(); return; }
  const arr = getByPath(state, group);
  const extraArr = extra ? getByPath(state, extra) : null;
  if(extraArr && !arr.includes(val) && !extraArr.includes(val)) extraArr.push(val);
  if(!arr.includes(val) && !(max && arr.length>=max)) arr.push(val);
  render();
}
function selectSingle(btn){ setByPath(state, btn.dataset.group, btn.dataset.value); render(); }
function toggleBool(btn){ const p = btn.dataset.path; setByPath(state, p, !getByPath(state,p)); render(); }
function addMockFile(btn){ const arr = getByPath(state, btn.dataset.path); arr.push(`${btn.dataset.prefix}_${arr.length+1}.jpg`); render(); }
function removeMockFile(btn){ const arr = getByPath(state, btn.dataset.path); arr.splice(parseInt(btn.dataset.index,10),1); render(); }

function chipGroup(opts){
  const selected = getByPath(state, opts.groupPath);
  const extraList = opts.extra ? getByPath(state, opts.extra) : [];
  const seen = new Set(); const list = [];
  [...opts.options, ...extraList].forEach(o=>{ if(!seen.has(o)){ seen.add(o); list.push(o); } });
  const atMax = !!(opts.max && selected.length>=opts.max);
  const chipsHtml = list.map(label=>{
    const on = selected.includes(label);
    if(!on && atMax){ return `<span class="chip chip--off">${escapeHtml(label)}</span>`; }
    return `<button type="button" class="chip ${on?'chip--on':''}" data-action="toggle-chip" data-group="${opts.groupPath}" data-value="${escapeAttr(label)}" data-max="${opts.max||''}">${on?'\u2713 ':''}${escapeHtml(label)}</button>`;
  }).join('');
  let customHtml = '';
  if(opts.custom){
    customHtml = `<div class="custom-row">
      <input type="text" id="${opts.customInputId}" placeholder="${escapeAttr(opts.placeholder||'Add your own\u2026')}" maxlength="28" />
      <button type="button" class="btn-add" data-action="add-custom" data-group="${opts.groupPath}" data-extra="${opts.extra||''}" data-input="${opts.customInputId}" data-max="${opts.max||''}">+ Add</button>
    </div>`;
  }
  const counter = opts.max ? `<div class="counter">${selected.length} / ${opts.max} selected</div>` : (selected.length ? `<div class="counter">${selected.length} selected</div>` : '');
  return `<div class="chips">${chipsHtml}</div>${customHtml}${counter}`;
}

function computeResult(){
  const tally = {comfort:0, explorer:0, health:0, trend:0};
  Object.keys(QUIZ_MAP).forEach(qk=>{
    const ans = state.a.quiz[qk]; const map = QUIZ_MAP[qk];
    if(ans && map[ans]) tally[map[ans]]++;
  });
  let best='explorer', bestScore=-1;
  Object.keys(tally).forEach(k=>{ if(tally[k]>bestScore){ best=k; bestScore=tally[k]; } });
  return PERSONALITY[best];
}

/* ========================= EVENTS ========================= */
document.getElementById('app').addEventListener('click', function(e){
  const btn = e.target.closest('[data-action]');
  if(!btn) return;
  switch(btn.dataset.action){
    case 'toggle-chip': toggleChip(btn); break;
    case 'add-custom': addCustom(btn); break;
    case 'select-single': selectSingle(btn); break;
    case 'toggle-bool': toggleBool(btn); break;
    case 'add-mock-file': addMockFile(btn); break;
    case 'remove-mock-file': removeMockFile(btn); break;
    case 'next': doNext(); break;
    case 'back': goBack(); break;
    case 'start': goTo('CREATE_ACCOUNT'); break;
    case 'restart': doRestart(); break;
    case 'noop': showNote('Log in isn\u2019t part of this onboarding demo.'); break;
  }
});
document.getElementById('app').addEventListener('input', function(e){
  const t = e.target;
  if(t.matches('input[data-field], textarea[data-field]')){
    setByPath(state, t.dataset.field, t.value);
  }
});
document.getElementById('app').addEventListener('keydown', function(e){
  if(e.key==='Enter' && e.target.matches('.custom-row input')){
    e.preventDefault();
    const row = e.target.closest('.custom-row');
    const addBtn = row.querySelector('[data-action="add-custom"]');
    if(addBtn) addCustom(addBtn);
  }
});

render();
