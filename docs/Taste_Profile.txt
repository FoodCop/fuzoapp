const TRACK_COLOR = '#F0DFCB';

/* ========================= STATE ========================= */
function blankTp(){
  const o = {};
  MODULES.forEach(m=>{
    o[m.key] = {};
    m.questions.forEach(q=>{ o[m.key][q.id] = q.type==='multi' ? [] : ''; });
  });
  return o;
}
function initialState(){ return { screen:'HUB', tp: blankTp() }; }
let state = initialState();

/* ========================= HELPERS ========================= */
function getByPath(obj, path){ return path.split('.').reduce((o,k)=> (o==null?o:o[k]), obj); }
function setByPath(obj, path, val){ const parts=path.split('.'); let o=obj; for(let i=0;i<parts.length-1;i++){ o=o[parts[i]]; } o[parts[parts.length-1]]=val; }
function escapeHtml(s){ return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }
function escapeAttr(s){ return String(s).replace(/&/g,'&amp;').replace(/"/g,'&quot;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }
function showNote(text){
  const t = document.createElement('div'); t.className='toast'; t.textContent=text;
  document.body.appendChild(t);
  setTimeout(()=>t.classList.add('toast--out'), 1700);
  setTimeout(()=>t.remove(), 2100);
}

/* ========================= COMPLETION LOGIC ========================= */
function questionAnswered(modKey, q){
  const val = state.tp[modKey][q.id];
  if(q.type==='multi'){
    if(q.requireExact) return Array.isArray(val) && val.length===q.max;
    return Array.isArray(val) && val.length>0;
  }
  return !!val;
}
function moduleAnsweredCount(mod){ return mod.questions.filter(q=>questionAnswered(mod.key,q)).length; }
function isModuleComplete(mod){ return moduleAnsweredCount(mod)===mod.questions.length; }
function modulePercent(mod){ return Math.round((moduleAnsweredCount(mod)/mod.questions.length)*100); }
function overallPercent(){ return MODULES.filter(isModuleComplete).length * 10; }
function overallFraction(){ return MODULES.filter(isModuleComplete).length / MODULES.length; }

/* ========================= INTERACTION HANDLERS ========================= */
function toggleChip(btn){
  const group = btn.dataset.group, value = btn.dataset.value;
  const maxAttr = btn.dataset.max; const max = maxAttr ? parseInt(maxAttr,10) : null;
  const arr = getByPath(state, group);
  const idx = arr.indexOf(value);
  if(idx>=0){ arr.splice(idx,1); }
  else{ if(max && arr.length>=max) return; arr.push(value); }
  render();
}
function selectSingle(btn){ setByPath(state, btn.dataset.group, btn.dataset.value); render(); }

/* ========================= RING SVG ========================= */
function ringSvg(fraction, size, stroke, color){
  const r = (size - stroke)/2;
  const c = 2*Math.PI*r;
  const offset = c*(1-Math.max(0,Math.min(1,fraction)));
  const center = size/2;
  return `<svg width="${size}" height="${size}" viewBox="0 0 ${size} ${size}">
    <circle cx="${center}" cy="${center}" r="${r}" stroke="${TRACK_COLOR}" stroke-width="${stroke}" fill="none" />
    <circle cx="${center}" cy="${center}" r="${r}" stroke="${color}" stroke-width="${stroke}" fill="none"
      stroke-dasharray="${c}" stroke-dashoffset="${offset}" stroke-linecap="round"
      transform="rotate(-90 ${center} ${center})" style="transition:stroke-dashoffset .4s ease" />
  </svg>`;
}

/* ========================= RENDER BITS ========================= */
function scaleControl(groupPath, value, accent){
  const opts = ['1','2','3','4','5'];
  return `<div class="scale-row">${opts.map(o=>`<button type="button" class="scale-btn ${value===o?'scale-btn--on':''}" data-action="select-single" data-group="${groupPath}" data-value="${o}">${o}</button>`).join('')}</div>
    <div class="scale-caption"><span>Not likely</span><span>Very likely</span></div>`;
}
function questionBlock(modKey, q, idx){
  let control = '';
  if(q.type==='single'){
    const val = state.tp[modKey][q.id];
    control = `<div class="chips">${q.options.map(o=>`<button type="button" class="chip ${val===o?'chip--on':''}" data-action="select-single" data-group="tp.${modKey}.${q.id}" data-value="${escapeAttr(o)}">${escapeHtml(o)}</button>`).join('')}</div>`;
  } else if(q.type==='multi'){
    const arr = state.tp[modKey][q.id];
    const atMax = arr.length>=q.max;
    const chipsHtml = q.options.map(o=>{
      const on = arr.includes(o);
      if(!on && atMax) return `<span class="chip chip--off">${escapeHtml(o)}</span>`;
      return `<button type="button" class="chip ${on?'chip--on':''}" data-action="toggle-chip" data-group="tp.${modKey}.${q.id}" data-value="${escapeAttr(o)}" data-max="${q.max}">${on?'\u2713 ':''}${escapeHtml(o)}</button>`;
    }).join('');
    const counterText = q.requireExact ? `${arr.length} of ${q.max} selected \u2014 pick exactly ${q.max}` : `${arr.length} / ${q.max} selected`;
    control = `<div class="chips">${chipsHtml}</div><div class="counter">${counterText}</div>`;
  } else if(q.type==='scale'){
    control = scaleControl(`tp.${modKey}.${q.id}`, state.tp[modKey][q.id]);
  }
  return `<div class="qblock">
    <div class="qblock__title">${idx+1}. ${escapeHtml(q.text)}</div>
    ${q.helper ? `<div class="qblock__helper">${escapeHtml(q.helper)}</div>` : ''}
    ${control}
  </div>`;
}

}

function screenModule(modId){
  const mod = MODULES[modId-1];
  const answered = moduleAnsweredCount(mod), total = mod.questions.length;
  const complete = answered===total;
  const p = modulePercent(mod);
  const body = mod.questions.map((q,i)=>questionBlock(mod.key,q,i)).join('');
  return `<div class="stage"><div class="menu" style="--mod-accent:${mod.accent}; --mod-accent-dark:${mod.accentDark}">
    <span class="brandmark">FUZO</span>
    <div class="tp-topbar">
      <button type="button" class="iconbtn iconbtn--text" data-action="to-hub">\u2190 Taste Profile</button>
      <span class="tp-topbar__badge">${complete ? 'Complete \u2713' : `${p}% done`}</span>
    </div>
    <div class="content">
      <h1 class="title">${mod.emoji} ${escapeHtml(mod.title)}</h1>
      <p class="subtitle">${escapeHtml(mod.blurb)} \u00b7 <strong>+10%</strong> to your taste profile when complete.</p>
      ${body}
    </div>
    <div class="nav">
      <button type="button" class="iconbtn" data-action="to-hub" aria-label="Back">\u2190</button>
      <button type="button" class="btn btn--primary" data-action="to-hub">Save & Return</button>
    </div>
  </div></div>`;
}

/* ========================= ROUTER ========================= */
function screenHtml(s){
  if(s==='HUB') return screenHub();
  if(s && s.indexOf('MOD_')===0) return screenModule(parseInt(s.split('_')[1],10));
  return screenHub();
}
function render(){ document.getElementById('app').innerHTML = screenHtml(state.screen); }

/* ========================= EVENTS ========================= */
document.getElementById('app').addEventListener('click', function(e){
  const btn = e.target.closest('[data-action]');
  if(!btn) return;
  switch(btn.dataset.action){
    case 'toggle-chip': toggleChip(btn); break;
    case 'select-single': selectSingle(btn); break;
    case 'open-module': state.screen = 'MOD_' + btn.dataset.mod; render(); window.scrollTo(0,0); break;
    case 'to-hub': state.screen = 'HUB'; render(); window.scrollTo(0,0); break;
    case 'reset-tp': state.tp = blankTp(); render(); break;
    case 'noop-profile': showNote('This would link back to the full profile page in the real app.'); break;
  }
});

render();
